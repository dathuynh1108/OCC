\begin{frame}[t]{PatchCore: build memory, then score new images}
\vspace{.48cm}
\begin{columns}[T,onlytextwidth]
\column{.40\textwidth}
\head{Build normal memory}
Collect normal patch descriptors.
\par\vspace{.28cm}
Greedy coreset: keep \textbf{10\%}.
\par\vspace{.28cm}
One memory per category.
\column{.54\textwidth}
\head{Get the score}
\textbf{Patch:} distance to the nearest memory patch.
\[
s_i=\min_{m\in\mathcal M}\|z_i-m\|_2.
\]
\textbf{Image:} take the largest patch score, then apply neighborhood reweighting.
\[
\begin{aligned}
s^*&=\max_i s_i,\\
s(x)&=\left(1-\frac{e^{\|m_{\mathrm{test},*}-m^*\|_2}}
{\sum_{m\in\mathcal N_b(m^*)}e^{\|m_{\mathrm{test},*}-m\|_2}}\right)s^*.
\end{aligned}
\]
\par\vspace{.28cm}
\textbf{Map:} arrange patch scores, upsample, smooth (Gaussian blur).
\end{columns}
\end{frame}

\begin{frame}[t]{PaDiM: a distributional view of patch anomalies}
\vspace{.45cm}
\begin{columns}[T,onlytextwidth]
\column{.42\textwidth}
\head{PatchCore / PatchScore}
Keep normal patches in a memory bank.
\par\vspace{.24cm}
New patch $\rightarrow$ nearest stored patch.
\column{.51\textwidth}
\head{PaDiM}
Fit one Gaussian per patch position.
\par\vspace{.24cm}
New patch $\rightarrow$ distance from its Gaussian.
\end{columns}
\vspace{.55cm}
\[
X_{ij}\sim\mathcal{N}(\mu_{ij},\Sigma_{ij}),\qquad
s_{ij}=\sqrt{(x_{ij}-\mu_{ij})^T\Sigma_{ij}^{-1}(x_{ij}-\mu_{ij})}.
\]
\vfill
{\scriptsize\textcolor{teal}{\href{https://arxiv.org/abs/2011.08785}{Defard et al., \textit{PaDiM: A Patch Distribution Modeling Framework for Anomaly Detection and Localization}, ICPR 2020.}}}
\end{frame}

\begin{frame}[t]{Kernel SVDD: enclose normal data in a sphere}
\small
\begin{columns}[T,onlytextwidth]
\column{.42\textwidth}
\centering\includegraphics[width=\linewidth,height=3.3cm,keepaspectratio]{svdd.pdf}
\raggedright
\head{Fit a compact boundary}
The kernel maps inputs into a space where normal observations lie near one center.
\column{.54\textwidth}
\[
\min_{a,R,\xi\ge0}R^2+\frac{1}{\nu n}\sum_i\xi_i,
\qquad\norm{\phi_k(x_i)-a}^2\le R^2+\xi_i.
\]
\[
s(x)=\norm{\phi_k(x)-a}^2-R^2.
\]
\head{Image benchmark in Ruff et al. (2018)}
Flatten images; fit PCA to retain 95\% of normal training variance. Use the Gaussian kernel:
\[
k(x,x')=\exp(-\gamma\norm{x-x'}_2^2).
\]
Search $\gamma\in\{2^{-10},\ldots,2^{-1}\}$ and $\nu\in\{0.01,0.1\}$.
\end{columns}
\end{frame}

\begin{frame}[t]{Deep SVDD: learn a compact normal representation}
\small
\begin{columns}[T,onlytextwidth]
\column{.44\textwidth}
\centering\includegraphics[width=\linewidth,height=2.55cm,keepaspectratio]{deep_svdd.pdf}
\raggedright
\head{Train the image encoder}
An autoencoder initializes the CNN. Deep SVDD then brings normal image embeddings toward a fixed center $c$.
\par\vspace{.18cm}
\textbf{MNIST:} LeNet, 8/4 channels, 32-D output.\\[.10cm]
\textbf{CIFAR-10:} LeNet, 32/64/128 channels, 128-D output.
\par\vspace{.18cm}
\head{Full reproduction schedule}
MNIST: AE 150 $\rightarrow$ SVDD 150 epochs.\\
CIFAR-10: AE 350 $\rightarrow$ SVDD 150 epochs.
\column{.52\textwidth}
\head{One-class objective}
\[
\min_\theta\frac1n\sum_i\norm{g_\theta(x_i)-c}^2
+\frac\lambda2\norm\theta^2.
\]
\head{Soft-boundary objective}
\[
\min_{\theta,R}\;R^2+\frac1{\nu n}\sum_i
\big[\norm{g_\theta(x_i)-c}^2-R^2\big]_+
+\frac\lambda2\norm\theta^2.
\]
\vspace{.22cm}
\head{Score a complete image}
\[
s(x)=\norm{g_\theta(x)-c}^2.
\]
For the soft boundary, subtract $R^2$; ranking is unchanged.
\end{columns}
\end{frame}

\def\phase{BASELINE METHODS}
\begin{frame}[t]{Deep SVDD: preventing collapse}
\small
\vspace{.14cm}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\head{The failure: one output for every input}
A constant mapping makes normal and anomalous observations indistinguishable.
\begin{alertblock}{Trivial solution}
\[
g_\theta(x)\equiv c\quad\Longrightarrow\quad s(x)=0.
\]
\end{alertblock}
\vspace{.14cm}
A trainable bias can produce this mapping:
\[
g_\theta(x)=Wh(x)+b,\qquad W=0,\;b=c.
\]
\column{.50\textwidth}
\head{Preventing collapse}
\textbf{Bias-free layers.} Remove the direct constant-output shortcut.
\par\vspace{.18cm}
\textbf{Fixed nonzero center.} Estimate $c$ from normal outputs and keep it fixed during training.
\par\vspace{.18cm}
\textbf{ReLU / leaky ReLU.} Avoid bounded activations with nonzero saturation.
\par\vspace{.18cm}
\head{Initialization}
Pretrain an autoencoder on normal images, then transfer its encoder weights.
\end{columns}
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{A local normality bubble}
\small
\[
B_j=(c_j,U_j,\Lambda_j,\sigma_{\perp,j}^2),\qquad
U_j^\top U_j=I_{d_j},\qquad\Pi_j=U_jU_j^\top.
\]
\begin{columns}[T,onlytextwidth]
\column{.57\textwidth}
\head{Local model}
\[
z=c_j+U_ja+\varepsilon_{\perp,j},\qquad a\perp\!\!\!\perp\varepsilon_{\perp,j},
\]
\[
a\sim\mathcal N(0,\Lambda_j),\qquad
\varepsilon_{\perp,j}\sim\mathcal N\!\left(0,\sigma_{\perp,j}^2(I-\Pi_j)\right).
\]
\[
\boxed{\Sigma_j=U_j\Lambda_jU_j^\top+\sigma_{\perp,j}^2(I-\Pi_j).}
\]
\column{.39\textwidth}
\centering\includegraphics[width=\linewidth,height=2.90cm,keepaspectratio]{bubble_detail.pdf}
\end{columns}
\vspace{0.10cm}
\head{Fit by local PCA}
Neighborhood mean $\to c_j$; top $d_j$ eigenpairs $\to U_j,\Lambda_j$.
\[
\sigma_{\perp,j}^2=\frac1{D-d_j}\sum_{k=d_j+1}^{D}\widehat\lambda_{jk}.
\]
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{Scoring a query}
\small
\[
a_j(z)=U_j^\top(z-c_j),\qquad
r_{\perp,j}(z)=(I-\Pi_j)(z-c_j).
\]
\begin{block}{Local energy}
\[
E_j(z)=
\underbrace{\frac{a_j^\top(\Lambda_j+\epsilon I)^{-1}a_j}{d_j}}_{\text{tangent rarity}}
+\beta\underbrace{\frac{\norm{r_{\perp,j}}_2^2}{(D-d_j)(\sigma_{\perp,j}^2+\epsilon)}}_{\text{off-manifold residual}}.
\]
\end{block}
\vspace{0.02cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\head{Absolute fit}
\[
S_{\mathrm{hard}}(z)=\min_{j\in\Cand(z)}E_j(z),
\]
\[
S_{\mathrm{soft}}(z)=-T_b\log\!\left[\frac1K\sum_{j\in\Cand(z)}e^{-E_j(z)/T_b}\right].
\]
\column{.48\textwidth}
\head{Attachment to the graph}
\[
p_j(z)=\frac{e^{-E_j(z)/T_b}}{\sum_{k\in\Cand(z)}e^{-E_k(z)/T_b}}.
\]
\end{columns}
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{Connect bubbles by three local criteria}
\small
\vspace{.16cm}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\head{1. Center compatibility}
\centering\includegraphics[width=\linewidth,height=1.75cm,keepaspectratio]{pair_centers.pdf}\par
\[
r_c(i,j)=\tfrac12\big[E_i(c_j)+E_j(c_i)\big].
\]
\raggedright
Each center should be plausible under the other bubble's local geometry.
\column{.31\textwidth}
\head{2. Tangent orientation}
\centering\includegraphics[width=\linewidth,height=1.75cm,keepaspectratio]{pair_frames.pdf}\par
\[
r_{\rm frame}(i,j)=\norm{\Pi_i-\Pi_j}_F^2
\]
\[
=d_i+d_j-2\norm{U_i^\top U_j}_F^2.
\]
\raggedright
Compare tangent subspaces, independent of basis-vector signs.
\column{.31\textwidth}
\head{3. Normal thickness}
\centering\includegraphics[width=\linewidth,height=1.75cm,keepaspectratio]{pair_thickness.pdf}\par
\[
r_{\perp}(i,j)=\left[\log\frac{\sigma_{\perp,i}^2+\epsilon}
{\sigma_{\perp,j}^2+\epsilon}\right]^2.
\]
\raggedright
Compare residual spread, not the full covariance.
\end{columns}
\vfill
\band{\small The three terms can determine graph edges. Diffusion then compares connectivity on this graph.}
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{From bubbles to a graph}
\small
\begin{block}{Edge dissimilarity, before diffusion}
\[
\delta_B(i,j)=\frac{r_c(i,j)}{\tau_c}
+\lambda_{\rm frame}\frac{r_{\rm frame}(i,j)}{\tau_{\rm frame}}
+\lambda_{\perp}\frac{r_{\perp}(i,j)}{\tau_{\perp}}.
\]
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\vspace{.20cm}
\includegraphics[width=\linewidth,height=2.85cm,keepaspectratio]{bubble_graph.pdf}
\column{.47\textwidth}
\head{Affinity and transition}
For symmetric local candidate edges $\mathcal E$,
\[
W_{ij}=\mathbf1_{\{(i,j)\in\mathcal E\}}e^{-\delta_B(i,j)}\quad(i\ne j),
\]
\[
W_{ii}=\eta>0,\qquad s_i=\sum_jW_{ij},
\]
\[
P_{ij}=\frac{W_{ij}}{s_i},\qquad\pi_i=\frac{s_i}{\sum_m s_m}.
\]
\end{columns}
\vspace{0.18cm}
\band{Fit the graph once on normal data. Every query uses the same bubble nodes.}
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{Diffusion distance}
\small
Similar $t$-step transition distributions give a smaller diffusion distance:
\[
\boxed{D_t^2(i,j)=\sum_m\frac{\big[(P^t)_{im}-(P^t)_{jm}\big]^2}{\pi_m}.}
\]
\begin{columns}[T,onlytextwidth]
\column{.44\textwidth}
\centering\vspace{0.12cm}
\includegraphics[width=\linewidth,height=2.67cm,keepaspectratio]{diffusion_walks.pdf}
\column{.52\textwidth}
\head{Compute with eigenvectors}
\[
P\psi_\ell=\lambda_\ell\psi_\ell,\qquad
\sum_i\pi_i\psi_\ell(i)\psi_k(i)=\delta_{\ell k}.
\]
\[
D_t^2(i,j)=\sum_{\ell=1}^{M-1}\lambda_\ell^{2t}\big[\psi_\ell(i)-\psi_\ell(j)\big]^2.
\]
\[
\Phi_t(B_j)=\big(\lambda_1^t\psi_1(j),\ldots,\lambda_L^t\psi_L(j)\big),
\]
\[
D_{t,L}^2(i,j)=\norm{\Phi_t(B_i)-\Phi_t(B_j)}^2.
\]
\end{columns}
\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{Does the query attach to one region?}
\small
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\includegraphics[width=.88\linewidth,height=1.28cm,keepaspectratio]{attachment_near.pdf}
\par\textbf{Nearby bubbles: lower dispersion}
\column{.48\textwidth}
\centering\includegraphics[width=.88\linewidth,height=1.28cm,keepaspectratio]{attachment_far.pdf}
\par\textbf{Distant bubbles: higher dispersion}
\end{columns}
\[
\boxed{S_D(z)=\frac1{2|\TT|}\sum_{t\in\TT}\sum_{i,j}p_i(z)p_j(z)D_{t,L}^2(i,j).}
\]
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\head{Equivalent variance at time $t$}
\[
\begin{aligned}
\bar\Phi_t(z)&=\sum_jp_j(z)\Phi_t(B_j),\\[0.2em]
S_{D,t}(z)&=\sum_jp_j(z)\norm{\Phi_t(B_j)-\bar\Phi_t(z)}^2.
\end{aligned}
\]
\column{.46\textwidth}
\head{Why keep a local score}
A sharp attachment can still have $D\approx0$, so the local score remains necessary.
\end{columns}

\end{frame}

\def\phase{NBD MODEL}
\begin{frame}[t]{Two final scores: hard min + diffusion and soft min + diffusion}
\small
\begin{block}{Combine local mismatch with diffusion dispersion}
\[
S_{\mathrm{soft}+D}(z)=\widetilde S_{\mathrm{soft}}(z)+\widetilde S_D(z),\qquad
S_{\mathrm{hard}+D}(z)=\widetilde S_{\mathrm{hard}}(z)+\widetilde S_D(z).
\]
\end{block}
\vspace{.12cm}
\head{Put the components on a common scale}
For each category, compare a score to its distribution on reserved normal patches:
\[
\widetilde S_q(z)=-\log\!\left[
\frac{1+\sum_{u\in\mathcal Z_{\rm cal}}\mathbf1_{\{S_q(u)\ge S_q(z)\}}}{|\mathcal Z_{\rm cal}|+1}
\right],\qquad q\in\{\mathrm{soft},\mathrm{hard},D\}.
\]
\vspace{.18cm}
\head{From patches to an image}
Average the 8 largest combined patch scores, using the same image aggregation as the feature-space controls.
\end{frame}

\def\phase{CONCLUSION}
\begin{frame}[t]{What the experiment shows}
\small
\vspace{.26cm}
\head{1. Frozen features set a high bar}
The ImageNet-pretrained Wide ResNet descriptor likely already captures much nonlinear normal structure: PatchScore with a normal memory bank reaches \textbf{98.57\% AUROC}.
\par\vspace{.30cm}
\head{2. Finding: compact centers retain most of PatchScore}
Replacing the \textbf{2,262}-vector PatchScore memory bank with only \textbf{128} bubble centers ($\sim$\textbf{17.7$\times$ fewer representatives}) still reaches \textbf{95.86\% AUROC}, just \textbf{2.71 pp} lower. The bubble construction therefore acts as a compact normal memory while preserving most of the useful structure.
\par\vspace{.30cm}
\head{3. The current limitation is diffusion}
Soft-min + diffusion loses \textbf{6.31 pp}; hard-min + diffusion loses \textbf{7.06 pp}. On these strong features, the evidence favors the local center score; next isolate graph connectivity and component scaling.
\end{frame}
